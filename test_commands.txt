# Basic set and get
SET key1 hello
GET key1

# Overwrite
SET key1 world
GET key1

# SETNX (set if not exists)
SETNX key1 fail
SETNX key2 success
GET key2

# MSET / MGET multiple keys
MSET key3 val3 key4 val4
MGET key1 key3 key4 key5   # key5 does not exist

# Append and length
APPEND key1 _appended
GET key1
STRLEN key1

# Increment / decrement
SET num 10
INCR num
GET num
INCRBY num 5
GET num
DECR num
DECRBY num 3
GET num

# Deletion
DEL key1 key2 key3 key4 num
GET key1
GET num


# Push and pop
LPUSH mylist a
LPUSH mylist b
RPUSH mylist c
LPRINT mylist

# Length and indexing
LLEN mylist
LINDEX mylist 0
LINDEX mylist 2

# LSET
LSET mylist 1 z
LPRINT mylist

# Pop operations
LPOP mylist
RPOP mylist
LPRINT mylist

# Sort (ascending)
LPUSH mylist 5
LPUSH mylist 2
LPUSH mylist 3
LSORT mylist ASC
LPRINT mylist
LSORT mylist DESC
LPRINT mylist




# Add members
SADD myset a b c
SMEMBERS myset
SCARD myset

# Check membership
SISMEMBER myset b
SISMEMBER myset x

# Remove member
SREM myset b
SMEMBERS myset

# Pop random member
SPOP myset
SMEMBERS myset

# Union, intersection, difference
SADD set1 1 2 3
SADD set2 2 3 4
SUNION set1 set2
SINTER set1 set2
SDIFF set1 set2


# HSET and HGET
HSET myhash field1 val1 field2 val2
HGET myhash field1
HEXISTS myhash field2
HLEN myhash

# Delete fields
HDEL myhash field1
HGET myhash field1
HLEN myhash




# Set key and expire
SET tempkey tempval
EXPIRE tempkey 5
TTL tempkey
GET tempkey

# Wait 6 seconds (simulate in tests or real sleep)
# After 6s:
TTL tempkey
GET tempkey




# Missing key
GET missingkey
DEL missingkey

# Wrong number of arguments
SET keyonly
MSET key1
INCR
LPOP
HGET myhash

# Unknown command
FOO key1
